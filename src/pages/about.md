### 为什么创建这个网站？

大一下在为建模比赛做准备，无意间看到一个论文模板的作者把他的个人博客也写到简介里了。为了解决一些问题，我进入他的博客寻找答案。他的博客页面新奇而又功能完备，给了我极大的震撼。博主在上面分享自己的生活，让我大开眼界，加上我这几年很少发朋友圈，于是萌生了做一个跟他类似网站的想法，作为一个**另类的朋友圈**。

当然我觉得做这个网站~~可以让我的代码能力有所提升~~!!（用来装逼十分合适）!!

我想借用这个博客分享一些我的生活日常，巩固记录我平日所学知识，希望这个博客能够在他人遇到困难之时能够提供一些**解决方案和思路**。

因为专业问题，以后我可能更多的发布一些代码方面的内容。但我深知这个专业是**壁垒极厚**的，因此我会尽力把文章写的通俗易懂，其中会用更多非专业的术语来解释，以达到让不了解计算机的人也能理解的程度，希望您能理解。

### 关于本站

创建这个站的时候，我想要的就是有一个能够**积累知识，积累兴趣**的地方，和他人分享，这样可以让知识成为真正的**积累和沉淀**。如果能够帮助到更多的人，解决更多问题，那一定是非常棒的事情。

这里大多是**技术向**的文章，可能不会有很多人看，就当是自我记录吧。一般我研究什么、发现了什么都会分享在这里。当然，如果某篇文章能够帮助到你，我也会感到很开心。

这些就是创建这个小站的本意，也是我**分享生活**的方式。有幸能和你相遇在这里，相信我们能共同留下一段**美好的记忆**。

<script>
  /**
   * 关于页的两处交互（旧站分别用 GSAP 与 jQuery 实现，这里都是原生写法）。
   *
   * `astro:page-load` 会在首屏与 ClientRouter 每次换页后触发，因此
   * 首屏直接调用一次、之后交给该事件，避免重复绑定。
   */
  function initHelloCard() {
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    for (const card of document.querySelectorAll('[data-hello-card]')) {
      if (card.dataset.helloBound === 'true') continue;
      card.dataset.helloBound = 'true';

      const cursor = card.querySelector('.about-cursor');
      const shapes = [...card.querySelectorAll('.about-shape')];
      if (!cursor || shapes.length === 0) continue;

      const factors = [0.18, 0.12, 0.07];
      let targetX = 0, targetY = 0, curX = 0, curY = 0, active = false;
      const shapeX = shapes.map(() => 0);
      const shapeY = shapes.map(() => 0);

      const tick = () => {
        curX += (targetX - curX) * 0.22;
        curY += (targetY - curY) * 0.22;
        cursor.style.transform = `translate3d(${curX}px, ${curY}px, 0)`;
        shapes.forEach((shape, i) => {
          const f = factors[i] ?? 0.1;
          const dx = active ? targetX * f : 0;
          const dy = active ? targetY * f : 0;
          shapeX[i] += (dx - shapeX[i]) * 0.12;
          shapeY[i] += (dy - shapeY[i]) * 0.12;
          shape.style.transform = `translate3d(${shapeX[i]}px, ${shapeY[i]}px, 0)`;
        });
        card.__helloFrame = requestAnimationFrame(tick);
      };

      card.addEventListener('mousemove', (event) => {
        const rect = card.getBoundingClientRect();
        targetX = event.clientX - rect.left - rect.width / 2;
        targetY = event.clientY - rect.top - rect.height / 2;
        active = true;
      });
      card.addEventListener('mouseleave', () => {
        active = false;
        targetX = 0;
        targetY = 0;
      });

      card.__helloFrame = requestAnimationFrame(tick);
    }
  }

  function initPursuitCard() {
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    for (const mask of document.querySelectorAll('[data-pursuit-words]')) {
      if (mask.dataset.pursuitBound === 'true') continue;
      mask.dataset.pursuitBound = 'true';

      const words = [...mask.querySelectorAll('.about-pursuit-word')];
      if (words.length < 2) continue;
      let index = 0;
      const apply = () => {
        words.forEach((w, i) => {
          w.classList.toggle('is-current', i === index);
          w.classList.toggle('is-previous', i === (index - 1 + words.length) % words.length);
        });
      };
      apply();
      setInterval(() => {
        index = (index + 1) % words.length;
        apply();
      }, 2000);
    }
  }

  function initAboutCards() {
    initHelloCard();
    initPursuitCard();
  }

  initAboutCards();
  document.addEventListener('astro:page-load', initAboutCards);
</script>
